# Raspberry Pi Zero / Zero W in the Retroflag GPi Case, Circle bare metal.
#
# The GPi Case normally needs Linux overlays and scripts (dpi24-gpi,
# pwm-audio, SafeShutdown_gpi.py). None of them run without Linux, so their
# settings are reproduced here with firmware "gpio=" lines, and again in the
# kernel (baremetal/platform/circle/circle_gpi.cpp).

kernel=kernel.img
gpu_mem=64
disable_splash=1
boot_delay=0

# --- Display: 2.8" DPI panel, natively 240x320 portrait, rotated to 320x240 ---
display_lcd_rotate=1
extra_transpose_buffer=2
enable_dpi_lcd=1
display_default_lcd=1
dpi_group=2
dpi_mode=87
dpi_output_format=0x6016
dpi_timings=240 1 38 10 20 320 1 20 4 4 0 0 0 60 0 6400000 1
framebuffer_width=320
framebuffer_height=240
framebuffer_depth=16
disable_overscan=1
overscan_left=0
overscan_right=0
overscan_top=0
overscan_bottom=0

# DPI pins (replaces dtoverlay=dpi24-gpi). Output format 6 (RGB666, mode 2)
# uses GPIO0-9, 12-17 and 20-25 and leaves 18/19 and 26/27 free.
gpio=0-17=a2,np
gpio=20-25=a2,np

# PWM audio on GPIO18 (left) and GPIO19 (right), ALT5 (replaces the
# pwm-audio overlay).
gpio=18-19=a5,np

# Soft power off (replaces SafeShutdown_gpi.py): GPIO27 holds the power latch
# of the case, GPIO26 reads the power switch (low = switched off).
# Takes effect with the case's "SAFE SHUTDOWN" switch set to ON.
gpio=27=op,dh
gpio=26=ip,pu

# GPIO14/15 are DPI data lines in the GPi Case, so the UART stays off.
enable_uart=0
